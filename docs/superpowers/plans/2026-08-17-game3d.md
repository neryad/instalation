**Goal:** Añadir una pantalla `/game3d` con raycasting 2.5D en primera persona (estilo retro) que reutiliza al 100% el motor narrativo actual (`engine/`), mostrando cada habitación como una celda 7x7 con puertas direccionales, silueta de la IA y panel de log inferior.

**Architecture:** Cada habitación se convierte en una escena de rejilla 7x7 (`engine3d/scene.ts`), renderizada por un raycaster DDA puro (`engine3d/raycaster.ts`) dibujado con `@shopify/react-native-skia`. El jugador (posición + dirección en la rejilla) se mueve por pasos de casilla y gira 90° (`engine3d/controls.ts`). Al llegar a una puerta aparece `DoorPrompt` con las opciones; elegir invoca `engine.move/forceDoor` (lógica intacta), una transición negra y la nueva habitación se construye desde el mismo `engine`. El panel narrativo inferior reutiliza `TerminalLog`.

**Tech Stack:** Expo SDK 54 / React Native 0.81, TypeScript strict, `@shopify/react-native-skia`, `vitest` (tests unitarios de TS puro).

```ts
import { PlayerState } from "../engine/player";
import { Direction, rooms } from "../engine/rooms";

export type Cell = 0 | 1 | 2; // 0 = suelo, 1 = muro, 2 = puerta
export type Grid = Cell[][];

export const GRID_SIZE = 7;
export const CENTER = (GRID_SIZE - 1) / 2; // 3

export interface DoorCell {
  direction: Direction;
  targetRoomId: string;
  locked: boolean; // requiere llave que el jugador no tiene
  forceable: boolean; // bloqueada y no inforzable
}

export interface RoomScene {
  roomId: string;
  grid: Grid;
  doors: Partial<Record<Direction, DoorCell>>;
}

const DIRS: Direction[] = ["norte", "sur", "este", "oeste"];
const EDGE = GRID_SIZE - 1;

// [col, fila] de la puerta en el centro de cada muro perimetral
const DOOR_COORDS: Record<Direction, [number, number]> = {
  norte: [CENTER, 0],
  sur: [CENTER, EDGE],
  este: [EDGE, CENTER],
  oeste: [0, CENTER],
};

export function effectiveConnections(
  roomId: string,
  sanity: number,
): Partial<Record<Direction, string>> {
  const room = rooms[roomId];
  let conns: Partial<Record<Direction, string>> = { ...room.connections };

  if (room.unstableConnections) {
    for (const u of room.unstableConnections) {
      if (sanity <= u.maxSanity) {
        conns = { ...conns, ...u.connections };
      }
    }
  }

  for (const d of DIRS) {
    const target = conns[d];
    if (!target) continue;
    const targetRoom = rooms[target];
    if (
      targetRoom.maxSanityToExist !== undefined &&
      sanity > targetRoom.maxSanityToExist
    ) {
      delete conns[d];
    }
  }

  return conns;
}

export function buildScene(state: PlayerState): RoomScene {
  const roomId = state.currentRoom;
  const conns = effectiveConnections(roomId, state.sanity);

  const grid: Grid = Array.from({ length: GRID_SIZE }, () =>
    Array(GRID_SIZE).fill(0) as Cell[],
  );

  for (let y = 0; y < GRID_SIZE; y++) {
    for (let x = 0; x < GRID_SIZE; x++) {
      const isEdge = y === 0 || y === GRID_SIZE - 1 || x === 0 || x === GRID_SIZE - 1;
      grid[y][x] = isEdge ? 1 : 0;
    }
  }

  const doors: RoomScene["doors"] = {};
  for (const d of DIRS) {
    const target = conns[d];
    if (!target) continue;
    const [x, y] = DOOR_COORDS[d];
    grid[y][x] = 2;
    const targetRoom = rooms[target];
    const hasKey = !!targetRoom.lockedBy && state.inventory.includes(targetRoom.lockedBy);
    doors[d] = {
      direction: d,
      targetRoomId: target,
      locked: !!targetRoom.lockedBy && !hasKey,
      forceable:
        !!targetRoom.lockedBy && !hasKey && !targetRoom.unbreakable,
    };
  }

  return { roomId, grid, doors };
}
```

```ts
import { Cell, Grid } from "./scene";

export interface RayColumn {
  perpDist: number;
  side: "N" | "S" | "E" | "W";
  type: Cell;
}

export interface RaycasterOptions {
  fovDeg?: number;
  numColumns?: number;
}

export const MAX_DIST = 100;

export function facingAngle(dir: "norte" | "sur" | "este" | "oeste"): number {
  switch (dir) {
    case "este":
      return 0;
    case "sur":
      return Math.PI / 2;
    case "oeste":
      return Math.PI;
    case "norte":
    default:
      return -Math.PI / 2;
  }
}

export function castRays(
  grid: Grid,
  px: number,
  py: number,
  angle: number,
  opts: RaycasterOptions = {},
): RayColumn[] {
  const numColumns = opts.numColumns ?? 120;
  const fov = ((opts.fovDeg ?? 66) * Math.PI) / 180;
  const halfTan = Math.tan(fov / 2);

  const dirX = Math.cos(angle);
  const dirY = Math.sin(angle);
  const planeX = -Math.sin(angle) * halfTan;
  const planeY = Math.cos(angle) * halfTan;

  const columns: RayColumn[] = [];

  for (let i = 0; i < numColumns; i++) {
    const cameraX = (2 * i) / numColumns - 1;
    const rayDirX = dirX + planeX * cameraX;
    const rayDirY = dirY + planeY * cameraX;

    let mapX = Math.floor(px);
    let mapY = Math.floor(py);
    const deltaX = rayDirX === 0 ? Infinity : Math.abs(1 / rayDirX);
    const deltaY = rayDirY === 0 ? Infinity : Math.abs(1 / rayDirY);

    let stepX: number;
    let sideDistX: number;
    if (rayDirX < 0) {
      stepX = -1;
      sideDistX = (px - mapX) * deltaX;
    } else {
      stepX = 1;
      sideDistX = (mapX + 1 - px) * deltaX;
    }

    let stepY: number;
    let sideDistY: number;
    if (rayDirY < 0) {
      stepY = -1;
      sideDistY = (py - mapY) * deltaY;
    } else {
      stepY = 1;
      sideDistY = (mapY + 1 - py) * deltaY;
    }

    let side: "N" | "S" | "E" | "W" = "N";
    let type: Cell = 1;
    let dist = MAX_DIST;
    const maxSteps = 64;

    for (let step = 0; step < maxSteps; step++) {
      let steppingX: boolean;
      if (sideDistX < sideDistY) {
        sideDistX += deltaX;
        mapX += stepX;
        steppingX = true;
      } else {
        sideDistY += deltaY;
        mapY += stepY;
        steppingX = false;
      }

      if (mapY < 0 || mapY >= grid.length || mapX < 0 || mapX >= grid[0].length) {
        dist = MAX_DIST;
        side = steppingX ? (rayDirX > 0 ? "W" : "E") : rayDirY > 0 ? "N" : "S";
        break;
      }

      const cell = grid[mapY][mapX];
      if (cell !== 0) {
        dist = steppingX ? sideDistX - deltaX : sideDistY - deltaY;
        type = cell;
        side = steppingX ? (rayDirX > 0 ? "E" : "W") : rayDirY > 0 ? "S" : "N";
        break;
      }
    }

    columns.push({ perpDist: dist, side, type });
  }

  return columns;
}
```

```ts
import { Direction } from "../engine/rooms";
import { CENTER, GRID_SIZE, RoomScene } from "./scene";

export interface Player3D {
  x: number; // columna en unidades de casilla (float)
  y: number; // fila en unidades de casilla (float)
  dir: Direction;
}

export type ForwardResult =
  | { kind: "move"; player: Player3D }
  | { kind: "door"; dir: Direction }
  | { kind: "blocked" };

export const TurnOrder: Direction[] = ["norte", "este", "sur", "oeste"];

const FACING: Record<Direction, { dx: number; dy: number }> = {
  norte: { dx: 0, dy: -1 },
  sur: { dx: 0, dy: 1 },
  este: { dx: 1, dy: 0 },
  oeste: { dx: -1, dy: 0 },
};

export function opposite(dir: Direction): Direction {
  const map: Record<Direction, Direction> = {
    norte: "sur",
    sur: "norte",
    este: "oeste",
    oeste: "este",
  };
  return map[dir];
}

export function turn(player: Player3D, left: boolean): Player3D {
  const idx = TurnOrder.indexOf(player.dir);
  const next = left ? (idx + 3) % 4 : (idx + 1) % 4;
  return { ...player, dir: TurnOrder[next] };
}

export function spawnPlayer(
  roomId: string,
  entryDir: Direction | null,
): Player3D {
  const cx = CENTER + 0.5;
  const cy = CENTER + 0.5;
  if (!entryDir) {
    return { x: cx, y: cy, dir: "norte" };
  }
  switch (entryDir) {
    case "norte":
      return { x: cx, y: 1.5, dir: "sur" };
    case "sur":
      return { x: cx, y: GRID_SIZE - 1.5, dir: "norte" };
    case "este":
      return { x: GRID_SIZE - 1.5, y: cy, dir: "oeste" };
    case "oeste":
    default:
      return { x: 1.5, y: cy, dir: "este" };
  }
}

export function forward(
  player: Player3D,
  scene: RoomScene,
): ForwardResult {
  const v = FACING[player.dir];
  const tx = player.x + v.dx;
  const ty = player.y + v.dy;
  const mx = Math.floor(tx);
  const my = Math.floor(ty);

  if (
    my < 0 ||
    my >= scene.grid.length ||
    mx < 0 ||
    mx >= scene.grid[0].length
  ) {
    return { kind: "blocked" };
  }

  const cell = scene.grid[my][mx];
  if (cell === 1) return { kind: "blocked" };
  if (cell === 2) return { kind: "door", dir: player.dir };

  return { kind: "move", player: { ...player, x: tx, y: ty } };
}
```

> Nota: `roomId` en `spawnPlayer` no se usa en v1 (la escena se deriva del estado del engine); se conserva el parámetro para futuros spawns por sala y para que la firma sea explícita.

```ts
import { PlayerState } from "../engine/player";

export interface SpriteProjection {
  visible: boolean;
  y1: number; // borde superior en px
  y2: number; // borde inferior en px
  halfWidth: number; // medio ancho en "columnas"
  xCenter: number; // columna central (centro de pantalla)
}

export interface SpriteParams {
  visible: boolean;
  distance: number;
}

const SPRITE_HEIGHT = 1.7; // unidades de mundo (más alto que una casilla)
const SPRITE_WIDTH = 0.7;

export function entitySpriteParams(state: PlayerState): SpriteParams {
  if (state.gameOver) return { visible: false, distance: 0 };
  if (state.entityRoom === state.currentRoom) {
    return { visible: true, distance: 1.6 };
  }
  if (state.entityAwareness > 70) {
    return { visible: true, distance: 4 };
  }
  return { visible: false, distance: 0 };
}

export function projectSprite(
  params: SpriteParams,
  numColumns: number,
  heightPx: number,
  fovDeg = 66,
): SpriteProjection {
  if (!params.visible || params.distance <= 0) {
    return { visible: false, y1: 0, y2: 0, halfWidth: 0, xCenter: 0 };
  }
  const projectionPlane =
    numColumns / 2 / Math.tan(((fovDeg / 2) * Math.PI) / 180);
  const screenHeight = (SPRITE_HEIGHT / params.distance) * projectionPlane;
  const screenWidth = (SPRITE_WIDTH / params.distance) * projectionPlane;
  const y1 = heightPx / 2 - screenHeight / 2;
  const y2 = heightPx / 2 + screenHeight / 2;
  return {
    visible: true,
    y1,
    y2,
    halfWidth: screenWidth / 2,
    xCenter: numColumns / 2,
  };
}
```

```tsx
import { Canvas, Rect } from "@shopify/react-native-skia";
import { useMemo } from "react";
import { facingAngle, Player3D } from "../../engine3d/controls";
import { castRays } from "../../engine3d/raycaster";
import { Grid } from "../../engine3d/scene";
import { projectSprite } from "../../engine3d/sprite";

const NUM_COLUMNS = 120;
const FOV = 66;

interface RaycastCanvasProps {
  grid: Grid;
  player: Player3D;
  entityDist: number;
  sanity: number;
  width: number;
  height: number;
}

function shade(hex: string, t: number): string {
  const r = Math.round(parseInt(hex.slice(1, 3), 16) * t);
  const g = Math.round(parseInt(hex.slice(3, 5), 16) * t);
  const b = Math.round(parseInt(hex.slice(5, 7), 16) * t);
  return `rgb(${r},${g},${b})`;
}

export function RaycastCanvas({
  grid,
  player,
  entityDist,
  sanity,
  width,
  height,
}: RaycastCanvasProps) {
  const columns = useMemo(
    () =>
      castRays(grid, player.x, player.y, facingAngle(player.dir), {
        numColumns: NUM_COLUMNS,
        fovDeg: FOV,
      }),
    [grid, player],
  );

  const sprite = useMemo(
    () =>
      projectSprite(
        { visible: entityDist > 0, distance: entityDist || 0 },
        NUM_COLUMNS,
        height,
        FOV,
      ),
    [entityDist, height],
  );

  const wallFactor = height / 2 / Math.tan(((FOV / 2) * Math.PI) / 180);
  const colW = width / NUM_COLUMNS;
  const isGlitch = sanity < 25;

  const rects: React.ReactNode[] = [];

  columns.forEach((c, i) => {
    const dist = Math.max(0.05, c.perpDist);
    const fog = Math.max(0.3, 1 - dist * 0.12);
    const wallH = Math.min(height * 2, wallFactor / dist);
    const startY = Math.max(0, (height - wallH) / 2);
    const endY = Math.min(height, (height + wallH) / 2);
    const x = i * colW - (isGlitch && i % 13 === 0 ? 3 : 0);

    const sideShade = c.side === "S" ? 0.7 : c.side === "N" ? 1 : 0.85;
    const base = c.type === 2 ? "#4a3a1a" : "#2a3a2a";
    const color = shade(base, fog * sideShade);

    rects.push(
      <Rect key={`ce-${i}`} x={x} y={0} width={colW + 1} height={startY} color="#050805" />,
      <Rect key={`wa-${i}`} x={x} y={startY} width={colW + 1} height={endY - startY} color={color} />,
      <Rect key={`fl-${i}`} x={x} y={endY} width={colW + 1} height={height - endY} color="#141a14" />,
    );
  });

  if (sprite.visible) {
    const sLeft = sprite.xCenter * colW - sprite.halfWidth * colW;
    rects.push(
      <Rect
        key="entity"
        x={sLeft}
        y={sprite.y1}
        width={sprite.halfWidth * 2 * colW}
        height={Math.max(1, sprite.y2 - sprite.y1)}
        color={isGlitch ? "#ff0000" : "#000000"}
      />,
    );
  }

  if (isGlitch) {
    rects.push(<Rect key="gl1" x={0} y={height * 0.25} width={width} height={4} color="#000" />);
    rects.push(<Rect key="gl2" x={0} y={height * 0.55} width={width} height={8} color="#000" />);
    rects.push(<Rect key="gl3" x={0} y={height * 0.8} width={width} height={3} color="#000" />);
  }

  return <Canvas style={{ width, height }}>{rects}</Canvas>;
}
```

```tsx
export interface ClassifiedDoors {
  open: DoorPromptOption[];
  locked: DoorPromptOption[];
  forceable: DoorPromptOption[];

  const open: DoorPromptOption[] = [];
  const locked: DoorPromptOption[] = [];

  return { open, locked, forceable };
```

```tsx
          hasSedative={state.inventory.includes("sedative")}
          disabled={state.gameOver || fading}
        />
      </View>

      <DoorPrompt
        visible={promptVisible && !fading}
        doors={doorOptions}
        onChoose={handleChoose}
        onForce={handleForce}
        onClose={() => setPromptVisible(false)}
      />
    </View>
  );
}
```
